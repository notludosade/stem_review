Below is a small TaskManager: add a task, look one up by id, and list them by priority. In a real project it would follow Lesson 23's advice and live in its own file, imported into a separate main. Read it slowly — nearly every unit in this course shows up somewhere in it.

Python 
class TaskNotFoundError(Exception):   # Unit 2: custom exception
    pass

class TaskManager:
    def __init__(self):
        self.tasks = {}   # Unit 3: hash map, O(1) lookup by id

    def add_task(self, task_id, description, priority):
        self.tasks[task_id] = (description, priority)

    def get_task(self, task_id):
        if task_id not in self.tasks:
            raise TaskNotFoundError(f"No task with id {task_id}")
        return self.tasks[task_id]

    def tasks_by_priority(self):
        return sorted(self.tasks.items(), key=lambda t: t[1][1])   # Unit 5: sort

manager = TaskManager()
manager.add_task(1, "Write report", 2)
manager.add_task(2, "Fix bug", 1)
for task_id, (description, priority) in manager.tasks_by_priority():
    print(priority, description)
# 1 Fix bug
# 2 Write report
Java 
class Task {   // Unit 1: a plain class holding related data
    String description;
    int priority;
    Task(String description, int priority) {
        this.description = description;
        this.priority = priority;
    }
}

class TaskNotFoundError extends RuntimeException {   // Unit 2
    TaskNotFoundError(String message) { super(message); }
}

class TaskManager {
    private Map<Integer, Task> tasks = new HashMap<>();   // Unit 3

    void addTask(int id, String description, int priority) {
        tasks.put(id, new Task(description, priority));
    }

    Task getTask(int id) {
        if (!tasks.containsKey(id)) {
            throw new TaskNotFoundError("No task with id " + id);
        }
        return tasks.get(id);
    }

    List<Task> tasksByPriority() {   // Unit 5: sort
        List<Task> sorted = new ArrayList<>(tasks.values());
        sorted.sort((a, b) -> a.priority - b.priority);
        return sorted;
    }
}

JavaScript 
class TaskNotFoundError extends Error {}   // Unit 2

class TaskManager {
    constructor() {
        this.tasks = new Map();   // Unit 3
    }

    addTask(id, description, priority) {
        this.tasks.set(id, { description, priority });
    }

    getTask(id) {
        if (!this.tasks.has(id)) {
            throw new TaskNotFoundError(`No task with id ${id}`);
        }
        return this.tasks.get(id);
    }

    tasksByPriority() {   // Unit 5: sort
        return [...this.tasks.entries()].sort((a, b) => a[1].priority - b[1].priority);
    }
}

const manager = new TaskManager();
manager.addTask(1, "Write report", 2);
manager.addTask(2, "Fix bug", 1);
for (const [id, task] of manager.tasksByPriority()) {
    console.log(task.priority, task.description);
}
// 1 "Fix bug"
// 2 "Write report"
C++ 
struct Task { std::string description; int priority; };   // Unit 1

class TaskNotFoundError : public std::runtime_error {   // Unit 2
public:
    explicit TaskNotFoundError(const std::string& msg) : std::runtime_error(msg) {}
};

class TaskManager {
public:
    void addTask(int id, const std::string& description, int priority) {
        tasks[id] = Task{description, priority};   // Unit 3
    }

    Task getTask(int id) {
        if (tasks.find(id) == tasks.end()) {
            throw TaskNotFoundError("No task with id " + std::to_string(id));
        }
        return tasks[id];
    }

    std::vector<std::pair<int, Task>> tasksByPriority() {   // Unit 5: sort
        std::vector<std::pair<int, Task>> sorted(tasks.begin(), tasks.end());
        std::sort(sorted.begin(), sorted.end(), [](const auto& a, const auto& b) {
            return a.second.priority < b.second.priority;
        });
        return sorted;
    }

private:
    std::unordered_map<int, Task> tasks;
};

int main() {
    auto manager = std::make_unique<TaskManager>();   // Unit 6: owned via a smart pointer
    manager->addTask(1, "Write report", 2);
    manager->addTask(2, "Fix bug", 1);
    for (const auto& [id, task] : manager->tasksByPriority()) {
        std::cout << task.priority << " " << task.description << std::endl;
    }
}

Trace the units through it: a Task class holds related data (Unit 1); TaskNotFoundError is a custom exception raised instead of returning something meaningless (Unit 2); the tasks themselves live in a hash map keyed by id for constant-time lookup, the same structure Unit 3 built by hand (Unit 3); tasksByPriority reaches for a sort exactly like Unit 5's, rather than writing bubble sort from scratch (Unit 5); and the C++ version wraps the whole manager in a unique_ptr, so its heap memory has one clear owner and frees itself automatically — Unit 6's ownership idea, applied (Unit 6). Two units don't appear directly in this code but belong in the exercise: try adding a totalEstimatedHours method that walks a task's subtasks recursively (Unit 4), and write a few unit tests for TaskManager the way Lesson 21 tested factorial — including one that checks get_task actually raises TaskNotFoundError for a missing id (Unit 7). If a test fails, Lesson 22's strategy — reproduce it, read the error, bisect — is exactly how to track it down.
